
const getProductsList = async (
  req,
  res
) => {
  try {
    const query =
      buildProductQuery(req);

    if (
      req.query.type ===
      "fastmoving"
    ) {

      const ids =
        await getFastMovingProductIds(
          req.query.shop
        );

      query._id = {
        $in: ids
      };
    }

    if (
      req.query.type ===
      "dormant"
    ) {

      const soldProducts =
        await getDormantProductIds(
          req.query.shop
        );

      query._id = {
        $nin: soldProducts
      };

      query.quantity = {
        $gt: 0
      };
    }

    const {
      type,
      sort,
      page,
      limit,
      reason = "all",
    } = req.query;


    const sortOptions =
      getSortOptions(
        sort,
        query
      );

    let productQuery =
      Product.find(query)
        .sort(sortOptions)
        .lean();

    if (page) {
      const pageInt =
        parseInt(page) || 1;

      const limitInt =
        parseInt(limit) || 10;

      const skip =
        (pageInt - 1) * limitInt;

      productQuery =
        productQuery
          .skip(skip)
          .limit(limitInt);
    }

    let products =
      await populateProductFields(
        productQuery
      );

    const inventories =
      await Inventory.find({
        shop: req.query.shop,
        product: {
          $in: products.map(p => p._id),
        },
      })
        .select("product quantity")
        .lean();

    const inventoryMap =
  await repairInventories({

    products,
    inventories,
    shopId: req.query.shop
  });

      
    

    products = products.map(
      product => ({
        ...product,
        quantity:
          inventoryMap[
          product._id.toString()
          ] || 0,
      })
    );

    if (
      page &&
      reason !== "download"
    ) {
      const totalProducts =
        await Product.countDocuments(
          query
        );

      return paginatedResponse({
        res,
        data: products,
        total: totalProducts,
        page,
        limit,
      });
    }

    return res.status(200).json(
      reason === "download"
        ? { data: products }
        : products
    );
  } catch (error) {
    console.error(error);

    return res.status(500).json({
      error:
        "Internal server error",
    });
  }
};
